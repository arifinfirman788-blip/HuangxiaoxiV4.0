import React, { useState, useEffect, useRef, useMemo } from 'react';

const NAV_ITEMS = [
  { id: 'Home', label: 'Home', iconSolid: <svg viewBox="0 0 24 24" fill="currentColor" className="w-6 h-6"><path d="M11.47 3.84a.75.75 0 011.06 0l8.69 8.69a.75.75 0 101.06-1.06l-8.689-8.69a2.25 2.25 0 00-3.182 0l-8.69 8.69a.75.75 0 001.061 1.06l8.69-8.69z" /><path d="M12 5.432l8.159 8.159c.03.03.06.058.091.086v6.198c0 1.035-.84 1.875-1.875 1.875H15a.75.75 0 01-.75-.75v-4.5a.75.75 0 00-.75-.75h-3a.75.75 0 00-.75.75V21a.75.75 0 01-.75.75H5.625a1.875 1.875 0 01-1.875-1.875v-6.198a2.29 2.29 0 00.091-.086L12 5.432z" /></svg>, iconOutline: <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" className="w-6 h-6"><path strokeLinecap="round" strokeLinejoin="round" d="M2.25 12l8.954-8.955c.44-.439 1.152-.439 1.591 0L21.75 12M4.5 9.75v10.125c0 .621.504 1.125 1.125 1.125H9.75v-4.875c0-.621.504-1.125 1.125-1.125h2.25c.621 0 1.125.504 1.125 1.125V21h4.125c.621 0 1.125-.504 1.125-1.125V9.75M8.25 21h8.25" /></svg> },
  { id: 'Search', label: 'Search', iconSolid: <svg viewBox="0 0 24 24" fill="currentColor" className="w-6 h-6"><path fillRule="evenodd" d="M10.5 3.75a6.75 6.75 0 100 13.5 6.75 6.75 0 000-13.5zM2.25 10.5a8.25 8.25 0 1114.59 5.28l4.69 4.69a.75.75 0 11-1.06 1.06l-4.69-4.69A8.25 8.25 0 012.25 10.5z" clipRule="evenodd" /></svg>, iconOutline: <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" className="w-6 h-6"><circle cx="11" cy="11" r="8"></circle><line x1="21" y1="21" x2="16.65" y2="16.65"></line></svg> },
  { id: 'Add', label: 'Create', iconSolid: <svg viewBox="0 0 24 24" fill="currentColor" className="w-6 h-6"><path fillRule="evenodd" d="M12 2.25c-5.385 0-9.75 4.365-9.75 9.75s4.365 9.75 9.75 9.75 9.75-4.365 9.75-9.75S17.385 2.25 12 2.25zM12.75 9a.75.75 0 00-1.5 0v2.25H9a.75.75 0 000 1.5h2.25V15a.75.75 0 001.5 0v-2.25H15a.75.75 0 000-1.5h-2.25V9z" clipRule="evenodd" /></svg>, iconOutline: <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" className="w-6 h-6"><line x1="12" y1="5" x2="12" y2="19"></line><line x1="5" y1="12" x2="19" y2="12"></line></svg> },
  { id: 'Heart', label: 'Likes', iconSolid: <svg viewBox="0 0 24 24" fill="currentColor" className="w-6 h-6"><path d="M11.645 20.91l-.007-.003-.022-.012a15.247 15.247 0 01-.383-.218 25.18 25.18 0 01-4.244-3.17C4.688 15.36 2.25 12.174 2.25 8.25 2.25 5.322 4.714 3 7.5 3c1.372 0 2.615.551 3.512 1.441.897-.89 2.14-1.441 3.512-1.441 2.786 0 5.25 2.322 5.25 5.25 0 3.924-2.438 7.111-4.739 9.256a25.175 25.175 0 01-4.244 3.17 15.247 15.247 0 01-.383.219l-.022.012-.007.004-.003.001a.752.752 0 01-.704 0l-.003-.001z" /></svg>, iconOutline: <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" className="w-6 h-6"><path d="M20.84 4.61a5.5 5.5 0 00-7.78 0L12 5.67l-1.06-1.06a5.5 5.5 0 00-7.78 7.78l1.06 1.06L12 21.23l7.78-7.78 1.06-1.06a5.5 5.5 0 000-7.78z"></path></svg> },
  { id: 'Profile', label: 'Me', iconSolid: <svg viewBox="0 0 24 24" fill="currentColor" className="w-6 h-6"><path fillRule="evenodd" d="M7.5 6a4.5 4.5 0 119 0 4.5 4.5 0 01-9 0zM3.751 20.105a8.25 8.25 0 0116.498 0 .75.75 0 01-.437.695A18.683 18.683 0 0112 22.5c-2.786 0-5.433-.608-7.812-1.7a.75.75 0 01-.437-.695z" clipRule="evenodd" /></svg>, iconOutline: <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" className="w-6 h-6"><path d="M20 21v-2a4 4 0 00-4-4H8a4 4 0 00-4 4v2"></path><circle cx="12" cy="7" r="4"></circle></svg> },
];

const LuminousFusion: React.FC = () => {
  const [activeTab, setActiveTab] = useState('Home');
  const [navWidth, setNavWidth] = useState(0);
  const [activeX, setActiveX] = useState(0);
  const [pillWidth, setPillWidth] = useState(0);
  const [isMoving, setIsMoving] = useState(false);
  const navRef = useRef<HTMLDivElement>(null);
  const moveTimeout = useRef<number | null>(null);

  useEffect(() => {
    const updateMetrics = () => {
      if (navRef.current) {
        const activeBtn = navRef.current.querySelector(`[data-id="${activeTab}"]`);
        if (activeBtn) {
          const rect = activeBtn.getBoundingClientRect();
          const parentRect = navRef.current.getBoundingClientRect();
          setNavWidth(parentRect.width);
          setActiveX(rect.left - parentRect.left + rect.width / 2);
          setPillWidth(rect.width);
        }
      }
    };

    updateMetrics();
    window.addEventListener('resize', updateMetrics);
    return () => window.removeEventListener('resize', updateMetrics);
  }, [activeTab]);

  const handleTabChange = (id: string) => {
    if (id === activeTab) return;
    setIsMoving(true);
    setActiveTab(id);
    if (moveTimeout.current) window.clearTimeout(moveTimeout.current);
    // 动效持续时间
    moveTimeout.current = window.setTimeout(() => setIsMoving(false), 800);
  };

  const dynamicPath = useMemo(() => {
    if (navWidth === 0) return "";
    
    const h = 120;
    const baseLineY = 15;
    const dipWidth = pillWidth + 45;
    const dipDepth = 85;
    
    const startX = activeX - dipWidth / 2;
    const endX = activeX + dipWidth / 2;

    return `
      M 0,${baseLineY}
      L ${Math.max(0, startX - 30)},${baseLineY}
      C ${startX - 5},${baseLineY} ${startX},${baseLineY + dipDepth} ${activeX},${baseLineY + dipDepth}
      C ${endX},${baseLineY + dipDepth} ${endX + 5},${baseLineY} ${Math.min(navWidth, endX + 30)},${baseLineY}
      L ${navWidth},${baseLineY}
      L ${navWidth},${h}
      L 0,${h}
      Z
    `;
  }, [navWidth, activeX, pillWidth]);

  return (
    <div className="absolute bottom-6 left-0 right-0 z-50 px-4 flex justify-center pointer-events-none">
      <div 
        ref={navRef}
        className="relative w-full max-w-[480px] h-[110px] flex items-end justify-between px-2 pb-2 overflow-visible pointer-events-auto"
      >
        {/* 背景层：极致毛玻璃与动态下沉路径 */}
        <div className="absolute inset-0 pointer-events-none overflow-visible">
          <svg 
            width="100%" 
            height="100%" 
            viewBox={`0 0 ${navWidth} 110`} 
            preserveAspectRatio="none"
          >
            <defs>
              <linearGradient id="glassEdge" x1="0" y1="0" x2="0" y2="1">
                <stop offset="0%" stopColor="white" stopOpacity="0.45" />
                <stop offset="40%" stopColor="white" stopOpacity="0.12" />
                <stop offset="100%" stopColor="white" stopOpacity="0" />
              </linearGradient>
            </defs>
            
            {/* 极高强度模糊背景 */}
            <path 
              d={dynamicPath} 
              fill="rgba(255, 255, 255, 0.085)" 
              className="transition-all duration-800 ease-[cubic-bezier(0.5,0,0,1)]"
              style={{ backdropFilter: 'blur(100px)', WebkitBackdropFilter: 'blur(100px)' }}
            />
            
            {/* 细节描边 */}
            <path 
              d={dynamicPath} 
              fill="none" 
              stroke="url(#glassEdge)" 
              strokeWidth="0.85"
              className="transition-all duration-800 ease-[cubic-bezier(0.5,0,0,1)]"
            />
          </svg>
        </div>

        {/* 交互胶囊：弹性拉伸动效 */}
        <div 
          className="absolute bg-white/20 backdrop-blur-[45px] border border-white/35 rounded-full transition-all duration-800 ease-[cubic-bezier(0.5,0,0,1)] shadow-[0_6px_35px_rgba(0,0,0,0.5)]"
          style={{
            left: `${activeX - (pillWidth * 0.82) / 2}px`,
            width: `${pillWidth * 0.82}px`,
            height: '46px',
            bottom: '24px',
            zIndex: 10,
            transform: isMoving ? 'scaleX(1.08) scaleY(0.96)' : 'scaleX(1) scaleY(1)',
          }}
        >
           <div className="absolute inset-0 rounded-full bg-gradient-to-br from-white/15 to-transparent pointer-events-none" />
           <div className="absolute inset-x-4 top-1 h-[1px] bg-white/20 blur-[1px]" />
        </div>

        {/* 导航按钮 */}
        {NAV_ITEMS.map((item) => {
          const isActive = activeTab === item.id;
          return (
            <button
              key={item.id}
              data-id={item.id}
              onClick={() => handleTabChange(item.id)}
              className={`relative z-20 h-[74px] flex items-center justify-center gap-2 px-6 rounded-full transition-all duration-500 group ${
                isActive ? 'text-white' : 'text-white/25 hover:text-white/50'
              }`}
            >
              <div className={`transition-all duration-700 ease-out ${isActive ? 'scale-105 translate-y-[-1px]' : 'scale-90 opacity-70 group-active:scale-80'}`}>
                {isActive ? item.iconSolid : item.iconOutline}
              </div>
              
              {isActive && (
                <span className="text-[12px] font-black tracking-tighter uppercase whitespace-nowrap overflow-hidden animate-in fade-in slide-in-from-left-2 duration-700">
                  {item.label}
                </span>
              )}
            </button>
          );
        })}

        {/* 底部微型导光条 */}
        <div 
          className="absolute bottom-1.5 h-[2.5px] bg-white/45 rounded-full transition-all duration-800 ease-[cubic-bezier(0.5,0,0,1)] shadow-[0_0_12px_rgba(255,255,255,0.4)]"
          style={{
            left: `${activeX - 10}px`,
            width: '20px',
            zIndex: 20
          }}
        />
      </div>
    </div>
  );
};

export default LuminousFusion;
